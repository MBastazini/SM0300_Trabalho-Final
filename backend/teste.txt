def preco_estimado(c, a0, k0, a, k, p, modelo="linear"):
    A = normaliza(a, a0)
    K = normaliza(k, k0)
    if modelo == "linear":
        P = c[0] + c[1]*A + c[2]*K
    else:
        P = c[0] + c[1]*A + c[2]*K + c[3]*A**2 + c[4]*A*K + c[5]*K**2
    return desnormaliza(p, P)
